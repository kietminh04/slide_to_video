    async function buildChaptersFromPages(pdf, totalPages, fileName) {
      const chapters = [];
      const scanLimit = Math.min(totalPages, 80);

      // Tính toán số chương linh hoạt theo độ dài tài liệu thực tế
      let numChapters = 5;
      if (totalPages <= 6) numChapters = Math.max(1, totalPages);
      else if (totalPages <= 14) numChapters = 3;
      else if (totalPages <= 28) numChapters = 4;
      else if (totalPages <= 50) numChapters = 5;
      else if (totalPages <= 80) numChapters = 6;
      else numChapters = Math.min(8, Math.round(totalPages / 10));

      const groupSize = Math.max(1, Math.ceil(scanLimit / numChapters));

      for (let chIdx = 0; chIdx < numChapters; chIdx++) {
        const startP = chIdx * groupSize + 1;
        if (startP > totalPages) break;
        const endP = Math.min(totalPages, (chIdx + 1) * groupSize);

        let chTitle = "";
        const candidateItems = [];

        // Đọc chọn lọc từ các trang trong phân đoạn này
        for (let p = startP; p <= endP && candidateItems.length < 4; p++) {
          try {
            const page = await pdf.getPage(p);
            const content = await page.getTextContent();
            const lines = [];
            let curLine = "";
            (content.items || []).forEach(it => {
              if (it.str && it.str.trim()) {
                curLine += (curLine ? " " : "") + it.str.trim();
                if (curLine.length > 50 || it.hasEOL) {
                  lines.push(curLine.trim());
                  curLine = "";
                }
              }
            });
            if (curLine && curLine.trim()) lines.push(curLine.trim());

            const meaningfulLines = lines.filter(l => {
              if (l.length < 5 || /^[0-9\s\-\.\/]+$/.test(l)) return false;
              if (/\b(springer|fienberg|olkin|new york|all rights reserved|isbn|contents|figure\s+\d+|slide\s+\d+|trang\s+\d+)\b/i.test(l)) return false;
              if (/^\d+\s+Chapter\s+\d+/i.test(l)) return false;
              return true;
            });

            if (!chTitle && meaningfulLines.length > 0) {
              chTitle = meaningfulLines[0].substring(0, 80).replace(/^(\d+[\.\-\:]|chapter|chương)\s*/i, '');
            }

            meaningfulLines.slice(1, 5).forEach(txt => {
              const cleanTxt = txt.substring(0, 90).trim();
              if (cleanTxt.length > 8 && !candidateItems.some(it => it.text === cleanTxt)) {
                // Nhận diện công thức toán học hoặc ký hiệu logic trong dòng
                let mathFormula = "";
                if (/[=<>\\\+\-\*\/\^\_\{\}\[\]\|\~]/.test(cleanTxt) || /\b(sum|int|lim|alpha|beta|theta|sigma|lambda|matrix|norm|loss|argmin|argmax)\b/i.test(cleanTxt)) {
                  mathFormula = `$${cleanTxt.replace(/\$/g, '')}$`;
                }
                candidateItems.push({
                  text: cleanTxt,
                  pdfPage: p,
                  formula: mathFormula
                });
              }
            });
          } catch (e) { }
        }

        if (!chTitle) {
          const docBase = (fileName || 'Bài giảng').replace(/\.[^/.]+$/, "");
          const fallbacks = [
            `Tổng quan & Đặt vấn đề nghiên cứu`,
            `Cơ sở lý thuyết & Mô hình hóa bài toán`,
            `Cơ chế thuật toán & Phương pháp tính toán`,
            `Đánh giá thực nghiệm & Phân tích tham số`,
            `Tổng kết phương pháp & Ứng dụng thực tiễn`,
            `Mở rộng bài toán & Hướng phát triển`
          ];
          chTitle = fallbacks[chIdx] || `Phân đoạn kiến thức ${chIdx + 1}`;
        }

        // Tạo cấu trúc chuẩn sư phạm: Mã số logic, không dùng 'Phần a' hay 'a1'
        const items = [];
        if (candidateItems.length > 0) {
          candidateItems.forEach((cItem, itIdx) => {
            items.push({
              code: `${chIdx + 1}.1.${itIdx + 1}`,
              text: cItem.text,
              duration: 30,
              pdfStart: cItem.pdfPage || startP,
              pdfEnd: cItem.pdfPage || endP,
              pageRange: `Slide ${cItem.pdfPage || startP}`,
              summary: `Phân tích khái niệm và ứng dụng trọng tâm: ${cItem.text}`,
              mechanism: cItem.formula || '',
              key_point: `Điểm then chốt của ${cItem.text}`,
              narration: '' // Để trống, người dùng hoặc hệ thống sẽ sinh
            });
          });
        } else {
          items.push(
            {
              code: `${chIdx + 1}.1.1`,
              text: `${chTitle} - Khái niệm & Cơ sở lý thuyết`,
              duration: 30,
              pdfStart: startP,
              pdfEnd: Math.min(totalPages, startP + 1),
              pageRange: `Slide ${startP} - ${Math.min(totalPages, startP + 1)}`,
              summary: `Trọng tâm cơ sở lý thuyết và định nghĩa chính của ${chTitle}.`,
              mechanism: '',
              key_point: `Nắm vững bản chất và điều kiện áp dụng của ${chTitle}.`,
              narration: ''
            },
            {
              code: `${chIdx + 1}.1.2`,
              text: `${chTitle} - Phương pháp & Cơ chế vận hành`,
              duration: 30,
              pdfStart: Math.min(totalPages, startP + 2),
              pdfEnd: endP,
              pageRange: `Slide ${Math.min(totalPages, startP + 2)} - ${endP}`,
              summary: `Phân tích phương pháp thực thi, giải thuật và các bước triển khai chi tiết.`,
              mechanism: '',
              key_point: `Quy trình và lưu ý quan trọng khi triển khai thực tế.`,
              narration: ''
            }
          );
        }

        chapters.push({
          chapter: `CHƯƠNG ${chIdx + 1}`,
          title: chTitle,
          depth: chIdx === 0 ? 'bridge' : (chIdx === numChapters - 1 ? 'bridge' : 'core'),
          visual_type: chIdx % 2 === 0 ? 'split_screen' : 'process_visualization',
          sections: [{
            code: `${chIdx + 1}.1`,
            title: chTitle,
            items: items
          }]
        });
      }

      return chapters;
    }
