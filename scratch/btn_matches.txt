      function fmtSrt(sec) {
        const sInt = Math.max(0, Math.floor(sec));
        const h = String(Math.floor(sInt / 3600)).padStart(2, '0');
        const m = String(Math.floor((sInt % 3600) / 60)).padStart(2, '0');
        const s = String(sInt % 60).padStart(2, '0');
        const ms = String(Math.floor((sec % 1) * 1000)).padStart(3, '0');
        return `${h}:${m}:${s},${ms}`;
      }

      let srtContent = '';

        const segStart = it.tStart;

        const segDur = Math.max(1, segEnd - segStart);

        if (!narr) {
          // Nếu chưa có lời thoại, xuất 1 cue đại diện cho phân cảnh
          srtContent += `${cueIndex++}\n${fmtSrt(segStart)} --> ${fmtSrt(segEnd)}\n[${it.code}] ${it.title}\n\n`;
          return;
        }

        // Tách câu thông minh để phụ đề không bị dồn 1 cục 200 từ
        const rawSentences = narr.split(/(?<=[.!?])\s+/).filter(s => s.trim().length > 0);
        const subChunks = [];

        rawSentences.forEach(sentence => {
          const words = sentence.trim().split(/\s+/);
          if (words.length <= 14) {
            subChunks.push(sentence.trim());
          } else {
            // Tách các câu quá dài thành các cụm 10-12 từ
            for (let i = 0; i < words.length; i += 10) {
              subChunks.push(words.slice(i, i + 10).join(' '));
            }
          }
        });

        if (subChunks.length === 0) {
          subChunks.push(narr);
        }

        const totalWords = subChunks.reduce((acc, c) => acc + c.split(/\s+/).length, 0);
        let currentSubStart = segStart;

        subChunks.forEach((chunk, cIdx) => {
          const chunkWords = chunk.split(/\s+/).length;
          const chunkShare = totalWords > 0 ? (chunkWords / totalWords) : (1 / subChunks.length);
          const chunkDur = (cIdx === subChunks.length - 1)
            ? Math.max(0.5, segEnd - currentSubStart)
            : Math.max(1.2, segDur * chunkShare);

          const currentSubEnd = Math.min(segEnd, currentSubStart + chunkDur);

          srtContent += `${cueIndex++}\n${fmtSrt(currentSubStart)} --> ${fmtSrt(currentSubEnd)}\n${chunk}\n\n`;
          currentSubStart = currentSubEnd;
