function getActiveScriptDocumentData() {
      if (!activeProject) return null;

      const allSources = activeProject.sources || [
        { name: (activeProject.title || 'Tài Liệu') + '.pdf', pages: 35, size: '1.2 MB', active: true }
      ];
      const activeSources = allSources.filter(s => s.active !== false);

      let cumulativeSec = 0;
      const chapters = [];
      const flatSegments = [];

      activeSources.forEach((src, srcIdx) => {
        const originalIdx = allSources.indexOf(src);
        const rootKey = getSourceKey(src, originalIdx >= 0 ? originalIdx : srcIdx);
        if (isNodeExcluded(rootKey)) return;

        const sourceScenes = (src.scenes && src.scenes.length > 0)
          ? src.scenes
          : (originalIdx === 0 ? currentScenes : []);
        const hierarchyChapters = getProjectHierarchy(activeProject, sourceScenes, src, originalIdx >= 0 ? originalIdx : srcIdx);

        hierarchyChapters.forEach((ch, chIdx) => {
          const chKey = `ch-${rootKey}-${chIdx}`;
          if (isNodeExcluded(chKey, [rootKey])) return;

          const chTitle = getNodeParam(chKey, 'title', ch.title || `Chương ${chIdx + 1}`);
          const chDepth = ch.depth || (chIdx === 0 ? 'bridge' : (chIdx === hierarchyChapters.length - 1 ? 'bridge' : 'core'));
          const chVisual = getNodeParam(chKey, 'visualType', ch.visual_type || 'split_screen');

          const chObj = {
            sourceName: src.name,
            sourceIdx: srcIdx,
            chKey,
            chIndex: chIdx + 1,
            title: chTitle,
            depth: chDepth,
            visualType: chVisual,
            sections: []
          };

          (ch.sections || []).forEach((sec, secIdx) => {
            const secKey = `sec-${rootKey}-${chIdx}-${secIdx}`;
            if (isNodeExcluded(secKey, [chKey, rootKey])) return;

            const secCode = sec.code || `${chIdx + 1}.${secIdx + 1}`;
            const secTitle = sec.title || `Phần ${secCode}`;

            const secObj = {
              secKey,
              code: secCode,
              title: secTitle,
              items: []
            };

            (sec.items || []).forEach((it, itIdx) => {
              const itKey = `item-${rootKey}-${chIdx}-${secIdx}-${itIdx}`;
              if (isNodeExcluded(itKey, [secKey, chKey, rootKey])) return;

              const itCode = getNodeParam(itKey, 'code', it.code || `${chIdx + 1}.${secIdx + 1}.${itIdx + 1}`);
              const itTitle = getNodeParam(itKey, 'title', it.text || it.title || `Mục ${itIdx + 1}`);
              const itDur = parseInt(getNodeParam(itKey, 'duration', it.duration || 30)) || 30;
              const itWpm = parseInt(getNodeParam(itKey, 'wpm', activeProject.wpm || 140)) || 140;
              const itVisual = getNodeParam(itKey, 'visualType', it.visual_type || chVisual || 'split_screen');
              const itNotes = getNodeParam(itKey, 'notes', '');
              const itMediaNote = getNodeParam(itKey, 'mediaNote', '');
              const itNarr = getNodeParam(itKey, 'narration', it.narration || '');
              const itAtts = getNodeParam(itKey, 'attachments', []);

              // Cơ chế công thức toán học và tóm tắt sư phạm
              const itMechanism = getNodeParam(itKey, 'mechanism', it.mechanism || '');
              const itSummary = getNodeParam(itKey, 'summary', it.summary || '');
              const itKeyPoint = getNodeParam(itKey, 'key_point', it.key_point || '');
              const itDetails = getNodeParam(itKey, 'details', it.details || '');
              const itExamples = getNodeParam(itKey, 'examples', it.examples || '');
              const itPageRange = it.pageRange || (it.pdfStart ? `Slide ${it.pdfStart} - ${it.pdfEnd}` : '');

              const tStart = cumulativeSec;
              const tEnd = cumulativeSec + itDur;
              cumulativeSec = tEnd;

              const cleanNarr = itNarr.trim();
              const wordCount = cleanNarr ? cleanNarr.split(/\s+/).length : 0;
              const estReadTime = wordCount > 0 ? Math.round((wordCount / itWpm) * 60) : 0;

              // Câu nối chuyển tiếp sư phạm (Storyboards)
              const bridge = getChapterBridge(itKey);

              const itemObj = {
                itKey,
                code: itCode,
                title: itTitle,
                duration: itDur,
                wpm: itWpm,
                tStart,
                tEnd,
                wordCount,
                estReadTime,
                visualType: itVisual,
                narration: itNarr,
                mechanism: itMechanism,
                summary: itSummary,
                keyPoint: itKeyPoint,
                details: itDetails,
                examples: itExamples,
                pageRange: itPageRange,
                notes: itNotes,
                mediaNote: itMediaNote,
                attachments: itAtts,
                bridge,
                chapterTitle: chTitle,
                sectionTitle: secTitle,
                sourceName: src.name
              };

              secObj.items.push(itemObj);
              flatSegments.push(itemObj);
            });

            if (secObj.items.length > 0) {
              chObj.sections.push(secObj);
            }
          });

          if (chObj.sections.length > 0) {
            chapters.push(chObj);
          }
        });
      });

      return {
        meta: {
          projectId: activeProject.id || currentProjectId,
          title: activeProject.title || 'Kịch Bản Bài Giảng',
          totalDuration: cumulativeSec,
          wpm: activeProject.wpm || 140,
          criticScore: parseInt(document.getElementById('score-overall')?.textContent) || 94,
          totalSources: activeSources.length,
          totalChapters: chapters.length,
          totalSegments: flatSegments.length,
          totalWords: flatSegments.reduce((sum, s) => sum + s.wordCount, 0),
          exportedAt: new Date().toLocaleString('vi-VN')
        },
        sources: activeSources.map(s => ({ name: s.name, pages: s.pages, size: s.size })),
        chapters,
        flatSegments
      };
    }

            if (it.mechanism) {
              lines.push(``, `> 📐 **Cơ chế & Công thức Toán học ($LaTeX$):**  `, `> ${it.mechanism}`);
            }

            if (it.summary || it.keyPoint) {
              lines.push(``, `> 💡 **Trọng tâm sư phạm & Điểm then chốt:**  `, `> ${it.summary || it.keyPoint}`);
            }

            if (it.bridge) {
              lines.push(``, `> 🌉 **Câu nối chuyển tiếp sư phạm (Bridge):**  `, `> *"${it.bridge}"*`);
            }

            lines.push(``, `##### 🎙️ Lời Thoại Sư Phạm:`);
            if (it.narration && it.narration.trim()) {
              lines.push(it.narration.trim());
            } else {
              lines.push(`*(Chưa tạo kịch bản lời thoại cho phân đoạn này)*`);
            }

            if (Array.isArray(it.attachments) && it.attachments.length > 0) {
              lines.push(``, `**📎 Tệp đính kèm minh họa:**`);
              it.attachments.forEach(att => {
                lines.push(`- 🖼️ \`${att.name}\` (${att.type} • ${att.size})`);
              });
            }

            lines.push(``, `---`, ``);
          });
        });
      });

      downloadFile(filename, lines.join('\n'), 'text/markdown');
      showToast(`Đã xuất tài liệu Markdown chuyên nghiệp: ${filename}`);
    });
